\section{Research Experience}
\cventry{2025--Present}{Transcriptomics of Friedreich Ataxia}{CSIR--Institute of Genomics and Integrative Biology}{Delhi}{}{
\begin{itemize}
\item Investigating transcriptomic alterations underlying \textbf{Friedreich Ataxia (FRDA)} using both patient blood RNA-seq datasets and induced pluripotent stem cell (iPSC)-derived cardiomyocyte models.
\item Performed differential expression and pathway enrichment analyses to identify blood-based transcriptomic biomarkers associated with disease severity, with the long-term objective of developing non-invasive molecular markers for patient stratification and disease monitoring.
\item Designed reproducible RNA-seq analysis pipelines involving quality assessment, normalization, batch-effect correction, differential expression analysis and functional enrichment using \textbf{Bioconductor, DESeq2, limma and custom R workflows}.
\item Currently investigating transcriptional remodeling during differentiation of \textbf{Day 15 and Day 52 iPSC-derived cardiomyocytes} from Friedreich Ataxia patients to understand molecular mechanisms contributing to progressive cardiomyopathy, the leading cause of mortality in FRDA.
\end{itemize}}
\vspace{0.5em}

\cventry{2024--2026}{Pharmacogenomics and Medical Relevance of Indian Populations}{CSIR--Institute of Genomics and Integrative Biology (GenomeIndia Project)}{Delhi}{}{
\begin{itemize}
\item Investigating the \textbf{landscape of clinically actionable pharmacogenomic variation} across the Indian population by integrating whole-genome sequencing data from \textbf{10,000 individuals representing 83 genetically diverse populations}.
\item Curated and harmonized pharmacogenomic evidence from \textbf{PharmGKB, CPIC, PharmVar, ClinPGx} and additional clinical resources to identify actionable SNPs, indels and star alleles relevant for precision medicine.
\item Characterized \textbf{population-specific pharmacogenomic variation}, quantified inter-population allele frequency differences, and compared Indian populations with global reference datasets to identify \textbf{ancestry-specific therapeutic implications}.
\item Performed comprehensive analyses of pharmacogenes including clinically important star alleles and metabolizer phenotypes to evaluate \textbf{differences in predicted drug response among Indian populations}.
\item Developed computational pipelines in \textbf{R, Python and Bash} for large-scale variant annotation, quality control, population-level analyses, visualization and statistical interpretation.
\end{itemize}}
\vspace{0.5em}

\cventry{2023--2024}{Master’s Thesis}{National Centre for Cell Science (NCCS)}{Pune}{}{
\begin{itemize}
\item Analyzed \textbf{viral genomes} to identify amino acid substitutions under positive selection, revealing sites of rapid evolution.
\item Constructed custom \textbf{substitution matrices} based on the PAM algorithm and compared them to standard \textbf{PAM matrices} to identify amino acid mutations under selection pressure.
\item Mapped key mutations onto protein structures, demonstrating a \textbf{non-random} distribution of substitutions localized to specific functional regions that facilitate the host-pathogen evolutionary arms race.
\end{itemize}}
\vspace{0.5em}

\cventry{2023}{Summer Research}{National Centre for Cell Science (NCCS)}{Pune}{}{
\begin{itemize}

\end{itemize}}
